# AUR packages only. Prefer packages/pacman.txt whenever an official package exists.
# One package per line; explain why it is necessary in a preceding comment.

# Password manager launched by the managed keybinds and its quick-access shortcut.
1password

# Direct HID control for the GZ302 rear lightbar; asusctl cannot address it separately.
# Installed by bootstrap only on a detected Z13.
z13ctl-bin

# greetd greeter in the DMS look; the official repositories only offer
# regreet, tuigreet, and agreety. Installed by scripts/setup-greetd.sh.
greetd-dms-greeter-bin

# Blip file transfer, opened on the chat workspace by a window rule; upstream
# publishes only an AUR package (blipnet), no official one.
blipnet

# Vicinae, the launcher and clipboard history (ADR-0024). Not in the
# official repositories; -bin repackages the upstream release instead of a
# long Qt/C++ and Node build. Its install script grants the snippet input
# server cap_dac_override and loads uinput for pasting.
vicinae-bin

# Handy offline push-to-talk dictation (ADR-0024). Upstream ships
# only release binaries and an AUR package; -bin repackages the upstream .deb
# instead of a Rust/Tauri and whisper build.
handy-bin
